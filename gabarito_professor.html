Solução em Python
# Implementação dinâmica segura
usuarios = ["Alice", "Bruno", "Carlos", "Diana", "Eva"]

def remover_usuario():
    nome_busca = input("Digite o nome do usuario a ser removido: ")
    if nome_busca in usuarios:
        usuarios.remove(nome_busca)
        print("Usuario removido com sucesso e estrutura reorganizada.")
    else:
        print("Erro: Usuario nao encontrado.")

def main():
    while True:
        print("\n--- MENU ---")
        print("1 - Cadastrar\n2 - Listar\n3 - Buscar\n4 - Atualizar\n5 - Remover\n6 - Sair")
        try:
            opcao = int(input("Escolha: "))
        except ValueError:
            print("Entrada invalida. Digite um numero.")
            continue
            
        if opcao == 5:
            remover_usuario()
        elif opcao == 2:
            for i, u in enumerate(usuarios):
                print(f"{i} - {u}")
        elif opcao == 6:
            break

if __name__ == "__main__":
    main()
